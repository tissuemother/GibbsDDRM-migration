cmake_minimum_required(VERSION 3.20)
project(gibbsddrm LANGUAGES CXX)
find_package(Torch REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")
add_library(gibbsddrm_core src/sampler.cpp)
target_link_libraries(gibbsddrm_core PUBLIC "${TORCH_LIBRARIES}")
target_include_directories(gibbsddrm_core PUBLIC src)
target_compile_options(gibbsddrm_core PRIVATE -Wall -Wextra)
add_executable(gibbsddrm src/main.cpp)
target_link_libraries(gibbsddrm PRIVATE gibbsddrm_core)
if(TARGET torch_cuda)
  target_compile_definitions(gibbsddrm PRIVATE GIBBSDDRM_CUDA=1)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(operator_check tests/operator_check.cpp)
  target_link_libraries(operator_check PRIVATE gibbsddrm_core)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME sampler_parity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/parity.py"
    --build "${CMAKE_CURRENT_BINARY_DIR}" --artifacts "${CMAKE_CURRENT_BINARY_DIR}/parity")
  add_test(NAME unet_export COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/export_check.py"
    --build "${CMAKE_CURRENT_BINARY_DIR}" --artifacts "${CMAKE_CURRENT_BINARY_DIR}/export")
  add_test(NAME benchmark_disabled COMMAND gibbsddrm --check-backend)
endif()
